### BBC Slave
#bno_set_bno_slave_hook
#bno_remove_bno_slave_hook
#bno_inherit_bno_slave
###

bno_set_bno_slave_hook = {
    $SOURCE$ = {
        if = {
            limit = {
                NOT = {
                    has_hook_of_type = {
                        type = bno_slave_hook
                        target = $TARGET$
                    }
                }
            }
            add_hook_no_toast = {
                type = bno_slave_hook
                target = $TARGET$
            }
        }
    }
}

bno_remove_bno_slave_hook = {
    $SOURCE$ = {
        if = {
            limit = {
                has_hook_of_type = {
                    type = bno_slave_hook
                    target = $TARGET$
                }
            }
            remove_hook = {
                type = bno_slave_hook
                target = $TARGET$
            }
        }
    }
}

bno_inherit_bno_slave = {
    $SOURCE$ = {
        remove_relation_slave = $TARGET$
        primary_heir = {
            set_relation_bno_slave = $TARGET$
        }
    }
}

remove_recipient_marriage_and_concubinage_effect = {
    scope:recipient = {
        if = {
            limit = {
                is_married = yes
                NOT = { is_spouse_of = scope:actor }
            }
            every_spouse = {
                divorce = scope:recipient
            }
        }
        if = {
            limit = {
                is_concubine = yes
                NOT = { is_concubine_of = scope:actor }
            }
            concubinist = {
                remove_concubine = scope:recipient
            }
        }
    }
}

bno_slave_interaction_invites_effect = {
    remove_recipient_marriage_and_concubinage_effect = yes
    scope:actor = {
        add_courtier = scope:recipient
    }
}

bno_slave_interaction_marries_effect = {
    remove_recipient_marriage_and_concubinage_effect = yes
    scope:actor = {
        marry = scope:recipient
    }
}
