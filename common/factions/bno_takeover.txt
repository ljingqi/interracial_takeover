@base_bnwo_discontent_progress = 5

bno_bbc_takeover_faction = {
    casus_belli = bno_bbc_takeover

    short_effect_desc = bno_bbc_takeover_faction.short_effect_desc

    sort_order = 999

    leaders_allowed_to_leave = no
    player_can_join = yes
    power_threshold = 0

    discontent_progress = {
        base = @base_bnwo_discontent_progress
        common_discontent_progress_modifier = yes
        #Culture
        modifier = {
            add = 10
            faction_leader = {
                save_temporary_scope_as = faction_leader
            }
            faction_target = {
                any_realm_county = {
                    culture = scope:faction_leader.culture
                }
            }
        }
        #Faith
        modifier = {
            add = 10
            faction_leader = {
                save_temporary_scope_as = faction_leader
            }
            faction_target = {
                any_realm_county = {
                    faith = scope:faction_leader.faith
                }
            }
        }
        #Not BBC/Slut with BBC Spouse
        modifier = {
            add = 20
            faction_target = {
                OR = {
                    is_bbc = no
                    trigger_if = {
                        limit = {
                            is_slut = yes
                        }
                        any_spouse = {
                            is_bbc = no
                        }
                    }
                }
            }
        }
        modifier = {
            add = -1000
            faction_target = {
                OR = {
                    is_bbc = yes
                    trigger_if = {
                        limit = {
                            is_slut = yes
                        }
                        any_spouse = {
                            is_bbc = yes
                            count = all
                        }
                    }
                }
            }
        }
    }

    name = bno_bbc_takeover_faction.name

    requires_county = yes
    requires_character = no

    is_valid = {
        always = yes
    }

    is_county_valid = {
        title_province = {
            geographical_region = world_africa
        }
        NOT = { culture = scope:faction.faction_leader.culture }
        NOT = { faith = scope:faction.faction_leader.faith }
    }

    is_character_valid = {
        is_culture_bnoc = yes
        faith = {
            is_considered_bno_faith = yes
        }
        OR = {
            is_bbc = yes
            trigger_if = {
                limit = {
                    is_slut = yes
                }
                any_spouse = {
                    is_bbc = yes
                    count = all
                }
            }
        }
    }

    demand = {
        save_scope_as = faction
        setup_bnwo_leader_effect = yes
        faction_leader = {
            add_opinion = {
                modifier = angry_opinion
                target = root.faction_target
                opinion = -100
            }
        }
        get_bnwo_revolt_target_effect = { FACTION = this }
        special_character = {
            save_scope_as = peasant_leader
        }
        special_title = {
            save_scope_as = target_title
        }
        save_temporary_scope_as = county_faction
        every_faction_county_member = {
            limit = {
                holder = {
                    NOT = { this = scope:faction.faction_target }
                    NAND = {
                        exists = joined_faction
                        joined_faction = {
                            this = scope:county_faction
                        }
                    }
                    is_vassal_of = scope:faction.faction_target
                }
            }
            holder = {
                add_to_list = bnwo_faction_vassal_targets
            }
        }
        every_faction_county_member = {
            holder = {
                every_liege_or_above = {
                    limit = {
                        NOT = { this = scope:faction.faction_target }
                        NAND = {
							exists = joined_faction
							joined_faction = {
								this = scope:county_faction
							}
						}
						is_vassal_of = scope:faction.faction_target
                        NOT = { is_in_list = bnwo_faction_vassal_targets }
                    }
                    add_to_list = bnwo_faction_vassal_targets
                }
            }
        }

        if = {
            limit = {
                any_in_list = {
                    list = bnwo_faction_vassal_targets
                    is_ai = no
                }
            }
            scope:faction = {
                set_variable = {
                    name = faction_targets_player
                    value = yes
                }
            }
        }
        #launch the demand event
        scope:faction.faction_target = {
            trigger_event = bno_embrace_african_1.0007
        }
    }

    can_character_join = {
        is_culture_bnoc = yes
        faith = {
            is_considered_bno_faith = yes
        }
        OR = {
            is_bbc = yes
            trigger_if = {
                limit = {
                    is_slut = yes
                }
                any_spouse = {
                    is_bbc = yes
                    count = all
                }
            }
        }
    }

    can_county_create = {
        county_opinion < 0
    }

    can_character_become_leader = {
        is_culture_bnoc = yes
        faith = {
            is_considered_bno_faith = yes
        }
        OR = {
            is_bbc = yes
            trigger_if = {
                limit = {
                    is_slut = yes
                }
                any_spouse = {
                    is_bbc = yes
                    count = all
                }
            }
        }
    }

    county_create_score = {
        base = 0
        modifier = {
            add = 100
            is_culture_bnoc = no
        }
        modifier = {
            add = 100
            faith = {
                is_considered_bno_faith = no
            }
        }
        modifier = {
            add = 100
            holder = {
                OR = {
                    is_bbc = no
                    trigger_if = {
                        limit = {
                            is_slut = yes
                        }
                        any_spouse = {
                            is_bbc = no
                        }
                    }
                }
            }
        }
        modifier = {
            factor = 0
            title_province = {
                NOT = { geographical_region = world_africa }
            }
        }
    }

    county_join_score = {
        base = 0
        modifier = {
            add = 100
            is_culture_bnoc = no
        }
        modifier = {
            add = 100
            faith = {
                is_considered_bno_faith = no
            }
        }
        modifier = {
            add = 100
            holder = {
                OR = {
                    is_bbc = no
                    trigger_if = {
                        limit = {
                            is_slut = yes
                        }
                        any_spouse = {
                            is_bbc = no
                        }
                    }
                }
            }
        }
        modifier = {
            factor = 0
            title_province = {
                NOT = { geographical_region = world_africa }
            }
        }
    }

    county_power = county_levies_to_raise

    ai_join_score = {
        base = 0
        modifier = {
            add = 100
            OR = {
                is_bbc = yes
                trigger_if = {
                    limit = {
                        is_slut = yes
                    }
                    any_spouse = {
                        is_bbc = yes
                        count = all
                    }
                }
            }
        }
    }

    on_creation = {
        set_variable = bno_bbc_takeover_faction
        random_faction_county_member = {
            save_scope_as = founding_county
        }
    }

    on_destroy = {
        set_variable = {
            name = peasant_destroying
            value = yes
        }
        if = {
            limit = { exists = special_character }
            special_character = {
                if = {
                    limit = { has_variable = peasant_title }
                    destroy_title = this.var:peasant_title
                }
                if = {
                    limit = {
                        is_alive = yes
                    }
                    remove_character_flag = bnwo_faction_claimant_without_title
                    remove_variable = peasant_title
                    remove_variable = bnwo_leader_peasants
                }
            }
        }
    }

    ai_demand_chance = {
        base = 100

        modifier = {
			add = -1000
			faction_target = {
				involved_activity ?= {
					has_activity_type = activity_coronation
					activity_host = prev
				}
			}
		}
    }

    on_war_start = {
        every_faction_county_member = {
            duchy = {
                add_to_list = areas_of_rebellion
            }
        }

        every_in_list = {
            list = areas_of_rebellion
            save_temporary_scope_as = this_duchy
            ordered_in_de_jure_hierarchy = {
                limit = {
                    tier = tier_county
                    any_title_joined_faction = {
                        this = root
                    }
                }

                order_by = total_county_levies

                title_province = {
                    save_temporary_scope_as = local_center_of_rebellion
                }
            }

            root.faction_leader = {
                spawn_bnwo_revolt_troops = yes
            }

            #give populist leader a commander
            create_character = {
                template = bbc_noble_template
                location = scope:local_center_of_rebellion
                save_scope_as = new_bnwo_commander
            }
            scope:new_bnwo_commander = {
                set_employer = root.faction_leader
            }
        }

        #give gold to leader
        every_faction_county_member = {
            root.faction_leader = {
                add_gold = {
                    value = minor_gold_value
                    add = 25
                }
            }
        }
    }

    leader_leaves = {
        #only trigger when leader got captured
        if = {
            limit = {
                special_character = {
                    is_alive = no
                }
            }
            faction_war = {
                end_war = defender
            }
            destroy_faction = yes
        }
        if = {
            limit = {
                NOT = { has_variable = peasant_destroying }
                exists = faction_war
            }
            faction_war = {
                end_war = defender
            }
        }
    }

    character_allow_create = no

    special_character_title = bnwo_faction_leader_title

    inherit_membership = yes

    multiple_targeting = yes
}
