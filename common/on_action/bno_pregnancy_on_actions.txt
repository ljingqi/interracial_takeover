on_pregnancy_mother = {
    on_actions = {
        bno_apply_lactation_story
        delay = { days = 7 }
        bno_on_pregnancy_mother
    }
}

bno_apply_lactation_story = {
    trigger = {
        scope:mother = {
            NOT = { owns_story_of_type = story_bno_lactation }
        }
    }
    effect = {
        scope:mother = {
            create_story = story_bno_lactation
        }
    }
}

bno_on_pregnancy_mother = {
    trigger = {
        scope:mother = {
            has_trait = bno_whitegirl
        }
    }
    effect = {
        scope:mother = {
            random = {
                chance = {
                    value = 0
                    switch = {
                        trigger = has_trait
                        bno_whitegirl_1 = { add = 20 }
                        bno_whitegirl_2 = { add = 40 }
                        bno_whitegirl_3 = { add = 60 }
                        bno_whitegirl_4 = { add = 80 }
                        bno_whitegirl_5 = { add = 100 }
                    }
                }
                remove_character_flag = birth_mother_will_die
                remove_character_flag = birth_child_will_become_sickly
                remove_character_flag = birth_mother_will_become_ill
                add_character_flag = birth_will_go_smoothly
            }
        }
    }
}
