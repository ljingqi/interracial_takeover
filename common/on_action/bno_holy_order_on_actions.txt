on_title_gain = {
    on_actions = {
        bno_on_title_gain
        bno_holy_order_on_title_gain
    }
}

on_title_gain_inheritance = {
    on_actions = {
        bno_on_title_gain
        bno_holy_order_on_title_gain
    }
}

on_title_gain_usurpation = {
    on_actions = {
        bno_on_title_gain
        bno_holy_order_on_title_gain
    }
}

bno_holy_order_on_title_gain = {
    trigger = {
        faith = {
            is_considered_bno_faith = yes
            any_faith_holy_order = {
                title = {
                    this = scope:title
                    NOT = { has_variable = bbc_holy_order }
                }
            }
        }
    }
    effect = {
        scope:title = {
            set_variable = bbc_holy_order
        }
        bno_make_holy_order_leader_blacked_girl_effect = yes
    }
}

bno_on_title_gain = {
    trigger = {
        scope:title = {
            has_variable = bbc_holy_order
        }
    }
    effect = {
        bno_make_holy_order_leader_blacked_girl_effect = yes
    }
}

on_join_court = {
    on_actions = {
        bno_on_join_court
        bno_holy_order_on_join_court
    }
}

bno_holy_order_on_join_court = {
    trigger = {
        save_temporary_scope_as = bno_new_courtier
        faith = {
            is_considered_bno_faith = yes
            any_faith_holy_order = {
                title = {
                    this = scope:bno_new_courtier.liege.primary_title
                    NOT = { has_variable = bbc_holy_order }
                }
            }
        }
    }
    effect = {
        liege.primary_title = {
            save_scope_as = title
            set_variable = bbc_holy_order
        }
        liege = {
            bno_make_holy_order_leader_blacked_girl_effect = yes
        }
    }
}

bno_on_join_court = {
    trigger = {
        liege.primary_title ?= {
            has_variable = bbc_holy_order
        }
    }
    effect = {
        bno_make_holy_order_member_blacked_girl_effect = yes
    }
}
